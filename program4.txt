//4. Program to Find the Biggest Number Among Three Numbers Using if-else
#include <stdio.h>
int main() {
    int a, b, c;
    printf("Enter three numbers: ");
    scanf("%d %d %d", &a, &b, &c);
    if (a >= b && a >= c) {
        printf("%d is the biggest number", a);
    }
    else if (b >= c) {
        printf("%d is the biggest number", b);
    }
    else {
        printf("%d is the biggest number", c);
    }
    return 0;
}
